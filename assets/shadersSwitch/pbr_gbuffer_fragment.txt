#version 300 es
precision highp float;
precision highp int;
precision highp sampler2D;

uniform vec3 DIFFUSE;
uniform vec3 SPECULAR;
uniform vec3 AMBIENT;
uniform sampler2D diffuseTexture;
uniform vec2 uvOffset;
uniform vec2 uvScale;
uniform float METALLIC;
uniform float ROUGHNESS;

in vec3 fragNormal;
in vec3 fragPosition;
in vec2 fragTexCoords;
in vec3 fragWorldPosition;

layout (location = 0) out vec3 gPosition;
layout (location = 1) out vec3 gNormal;
layout (location = 2) out vec4 gAlbedoMetallic;
layout (location = 3) out float gRoughness;

void main() {
    gPosition = fragWorldPosition;
    gNormal = normalize(fragNormal);
    
    vec2 transformedUV = fragTexCoords * uvScale + uvOffset;
    vec4 texColor = texture(diffuseTexture, transformedUV);
    
    if (texColor.a < 0.1) {
        discard;
    }

    vec3 albedo = DIFFUSE * texColor.rgb;
    
    gAlbedoMetallic = vec4(albedo, METALLIC);
    gRoughness = ROUGHNESS;
}
